class Duplicates {
    public static void main(String[] args) {
        int[] arr = {1, 2, 2, 3, 1, 4};
        
        for (int i = 0; i < arr.length; i++) {
            int count = 1;
            if (arr[i] != -1) {
                for (int j = i + 1; j < arr.length; j++) {
                    if (arr[i] == arr[j]) {
                        count++;
                        arr[j] = -1; // Mark as counted
                    }
                }

            }
        }
    }
}

class MergeSorted {

Q2Implement Stack Using Array
LIFO: Last In, First Out (using `top` pointer) 
class SimpleStack {
    int[] stack = new int[5];
    int top = -1;

    void push(int data) {
        if (top < 4) stack[++top] = data; // Add to top
    }
    
    void pop() {
        if (top >= 0) top--; // Remove from top
    }
    
    public static void main(String[] args) {
        SimpleStack s = new SimpleStack();
        s.push(10); 
        s.push(20);
        s.pop(); // Removes 20
        System.out.println("Top element index: " + s.top);
    }
}
Q3Implement Queue Using Array
FIFO: First In, First Out (using `front` and `rear` pointers) 
class SimpleQueue {
    int[] q = new int[5];
    int front = 0, rear = 0;

    void enqueue(int data) {
        if (rear < 5) q[rear++] = data; // Add to rear
    }
    
    void dequeue() {
        if (front < rear) front++; // Remove from front
    }
    
    public static void main(String[] args) {
        SimpleQueue sq = new SimpleQueue();
        sq.enqueue(10); 
        sq.enqueue(20);
        sq.dequeue(); // Removes 10
    }
}

class NumberObj {
    int value;
    
    NumberObj(int v) { value = v; }
    
    // Parameter is an object, Return type is an object
    NumberObj addObject(NumberObj obj2) {
        return new NumberObj(this.value + obj2.value); 
    }
    
    public static void main(String[] args) {
        NumberObj n1 = new NumberObj(10);
        NumberObj n2 = new NumberObj(20);
        
        NumberObj result = n1.addObject(n2);

    }

Q14For Loop Patterns
class Patterns {
    public static void main(String[] args) {
        // Pattern (a): Number Triangle
        for (int i = 1; i <= 5; i++) {
            for (int j = 1; j <= i; j++) System.out.print(i);
            System.out.println();
        }
        
        // Pattern (b): Inverted 'S' Triangle
        for (int i = 5; i >= 1; i--) {
            for (int j = 1; j <= i; j++) System.out.print("S");
            System.out.println();
        }
        
        // Pattern (c): Right-Aligned Number Triangle
        for (int i = 1; i <= 5; i++) {
            for (int space = 5; space > i; space--) System.out.print(" ");
            for (int j = 1; j <= i; j++) System.out.print(i);
            System.out.println();
        }
    }
}

class EulerFormula {
    public static void main(String[] args) {
        double e = 1, fact = 1;
        
        for (int i = 1; i <= 10; i++) { // Looping to 10 for precision

    }
