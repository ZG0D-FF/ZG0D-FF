/* ════════════════════════════════════════ THREE.JS — 3D JARVIS HOLOGRAM (PERFORMANCE SCALED) ════════════════════════════════════════ */ function init3D() { const canvas = document.getElementById('three-canvas'); // 1. HARDWARE DETECTION TRICK // Check CPU cores to determine if this is a high-end or low-end device const logicalCores = navigator.hardwareConcurrency || 4; const isHighEnd = logicalCores >= 8; // Scale variables based on device power const pCount = isHighEnd ? 300 : 80; // Way fewer particles for mobile const ringCount = isHighEnd ? 9 : 4; // Fewer 3D rings for mobile const pixelRatio = isHighEnd ? Math.min(window.devicePixelRatio, 2) : 1; // Lower resolution on mobile const renderer = new THREE.WebGLRenderer({ canvas, antialias: isHighEnd, alpha: true }); renderer.setPixelRatio(pixelRatio); renderer.setSize(window.innerWidth, window.innerHeight); renderer.setClearColor(0x000000, 0); const scene = new THREE.Scene(); const camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000); camera.position.set(0, 0, 5); const C = 0x00e5ff; /* ── WIREFRAME SPHERE (core) ── */ const sphereGeo = new THREE.SphereGeometry(1.1, 16, 12); const sphereMat = new THREE.MeshBasicMaterial({ color: C, wireframe: true, transparent: true, opacity: 0.08 }); const sphere = new THREE.Mesh(sphereGeo, sphereMat); scene.add(sphere); /* ── ICOSAHEDRON (inner core pulse) ── */ const icoGeo = new THREE.IcosahedronGeometry(0.55, 1); const icoMat = new THREE.MeshBasicMaterial({ color: C, wireframe: true, transparent: true, opacity: 0.25 }); const ico = new THREE.Mesh(icoGeo, icoMat); scene.add(ico); /* ── TORUS RINGS (SCALED) ── */ function makeTorus(r, tube, tilt, speed, opacity) { // Lower segment count for better performance const geo = new THREE.TorusGeometry(r, tube, isHighEnd ? 6 : 4, isHighEnd ? 80 : 40); const mat = new THREE.MeshBasicMaterial({ color: C, transparent: true, opacity }); const mesh = new THREE.Mesh(geo, mat); mesh.rotation.x = tilt; mesh.userData.speed = speed; return mesh; } const allRings = [ makeTorus(1.55, 0.008, 0, 0.004, 0.7), makeTorus(1.55, 0.004, Math.PI/2, -0.006, 0.4), makeTorus(1.8, 0.006, Math.PI/6, -0.003, 0.5), makeTorus(2.1, 0.005, Math.PI/2, -0.002, 0.35), // Extra rings only for high-end devices makeTorus(1.55, 0.004, Math.PI/4, 0.005, 0.3), makeTorus(1.8, 0.004, -Math.PI/3, 0.007, 0.25), makeTorus(2.1, 0.003, Math.PI/5, 0.004, 0.2), makeTorus(2.5, 0.006, 0, -0.0015, 0.15), makeTorus(2.5, 0.003, Math.PI/3, 0.003, 0.1) ]; const rings = allRings.slice(0, ringCount); rings.forEach(r => scene.add(r)); /* ── PARTICLES (SCALED) ── */ const pPositions = new Float32Array(pCount * 3); const pVel = []; for (let i = 0; i < pCount; i++) { const r = 2.8 + Math.random() * 2.5; const theta = Math.random() * Math.PI * 2; const phi = Math.acos(2 * Math.random() - 1); pPositions[i*3] = r * Math.sin(phi) * Math.cos(theta); pPositions[i*3+1] = r * Math.sin(phi) * Math.sin(theta); pPositions[i*3+2] = r * Math.cos(phi); pVel.push({ vx: (Math.random()-.5)*0.003, vy: (Math.random()-.5)*0.003, vz: (Math.random()-.5)*0.003 }); } const pGeo = new THREE.BufferGeometry(); pGeo.setAttribute('position', new THREE.BufferAttribute(pPositions, 3)); const pMat = new THREE.PointsMaterial({ color: C, size: 0.025, transparent: true, opacity: 0.6 }); const points = new THREE.Points(pGeo, pMat); scene.add(points); /* ── MOUSE PARALLAX ── */ let mouseX = 0, mouseY = 0; document.addEventListener('mousemove', e => { mouseX = (e.clientX / window.innerWidth - 0.5) * 2; mouseY = (e.clientY / window.innerHeight - 0.5) * 2; }); /* ── RESIZE ── */ window.addEventListener('resize', () => { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); }); /* ── ANIMATE ── */ let t = 0; function animate() { requestAnimationFrame(animate); t += 0.01; scene.rotation.y += (mouseX * 0.3 - scene.rotation.y) * 0.02; scene.rotation.x += (-mouseY * 0.2 - scene.rotation.x) * 0.02; rings.forEach(r => { r.rotation.z += r.userData.speed; r.rotation.y += r.userData.speed * 0.3; }); sphere.rotation.y += 0.001; sphere.rotation.x += 0.0005; ico.rotation.y -= 0.008; ico.rotation.x += 0.005; ico.scale.setScalar(1 + Math.sin(t * 1.5) * 0.06); const pos = pGeo.attributes.position.array; for (let i = 0; i < pCount; i++) { pos[i*3] += pVel[i].vx; pos[i*3+1] += pVel[i].vy; pos[i*3+2] += pVel[i].vz; const dist = Math.sqrt(pos[i*3]**2 + pos[i*3+1]**2 + pos[i*3+2]**2); if (dist > 6 || dist < 2.5) { pVel[i].vx *= -1; pVel[i].vy *= -1; pVel[i].vz *= -1; } } pGeo.attributes.position.needsUpdate = true; renderer.render(scene, camera); } animate(); }
